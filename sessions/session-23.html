Step 2 — Classroom (aggregates Students, associated with Teacher)

// File: Classroom.java
package schoolsystem;


import java.util.*;
class Classroom
{
private String roomNumber; // ASSOCIATION end: the room knows its teacher (may be null = unstaffed)
public String getRoomNumber() { return roomNumber; }
private void setRoomNumber(String roomNumber) { this.roomNumber = roomNumber; }
private Teacher teacher; // AGGREGATION: students are handed in from outside, list kept private
public Teacher getTeacher() { return teacher; }
void setTeacher(Teacher teacher) { this.teacher = teacher; }
private List<Student> students = new ArrayList<>();

public int getStudentCount() { return students.size(); }

public Classroom(String roomNumber)
{
    this.roomNumber = roomNumber;
}

// Called by teacher.AssignTo - keeps BOTH ends of the link consistent
public void SetTeacher(teacher t)
{
    this.teacher = t;
}

public void enroll(Student s)
{
    if (students.contains(s))
    {
        System.out.println(s.getName() + " is already in " + roomNumber + ".");
        return;
    }
    students.add(s);
    System.out.println(s.getName() + " enrolled in " + roomNumber + ".");
}

public void remove(Student s)
{
    students.remove(s);
}

public void transfer(Student s, Classroom target)
{
    if (!students.contains(s))
    {
        System.out.println(s.getName() + " is not in " + roomNumber + " - cannot transfer.");
        return;
    }
    remove(s);
    target.enroll(s);
    System.out.println(s.getName() + " transferred " + roomNumber + " → " + target.getRoomNumber() + ".");
}

public void printRegister()
{
    String teacherName = this.teacher == null ? "(no teacher)" : teacher.getName();
    System.out.println("--- Room " + roomNumber + " | teacher: " + teacherName + " | " + getStudentCount() + " students ---");
    for (Student s : students)
    {
        System.out.println("   " + s);
    }
}
}

Step 3 — Teacher (uses Classroom — association)

// File: teacher.java
package schoolsystem;

class Teacher
{
private String name;
public String getName() { return name; }
private void setName(String name) { this.name = name; }
private String subject; // ASSOCIATION: a reference to an independently-living Classroom (or null)
public String getSubject() { return subject; }
private void setSubject(String subject) { this.subject = subject; }
private Classroom assignedRoom;
public Classroom getAssignedRoom() { return assignedRoom; }
private void setAssignedRoom(Classroom assignedRoom) { this.assignedRoom = assignedRoom; }
public Teacher(String name, String subject)
{
    this.name = name;
    this.subject = subject;
}

public void assignTo(Classroom room)
{
    // Business rule: a classroom can have at most ONE teacher
    if (room.getTeacher() != null && room.getTeacher() != this)
    {
        System.out.println("Room " + room.getRoomNumber() + " already has " + room.getTeacher().getName() + "!");
        return;
    }

    // If this teacher had a previous room, unlink it first
    if (assignedRoom != null)
    {
        assignedRoom.SetTeacher(null);
    }

    this.assignedRoom = room;
    room.SetTeacher(this);
    System.out.println(name + " assigned to room " + room.getRoomNumber() + ".");
}

public void teach()
{
    if (this.assignedRoom == null)
    {
        System.out.println(name + " has no classroom.");
        return;
    }
    System.out.println(name + " is teaching " + subject + " in " + assignedRoom.getRoomNumber() + " " +
                      "to " + assignedRoom.getStudentCount() + " students.");
}
}
🧠 Beginner's Logic Breakdown
  1. A two-way association needs both ends updated: Teacher points at the room AND the room points back. AssignTo is the single method responsible for keeping both references consistent.
  2. Guard clauses defend relationship rules: "at most one teacher per room" is coded as the room.Teacher != null check before linking.
  3. Aggregation surface: the students list is private; outsiders use Enroll/Remove/TransferTo. Encapsulation (Lesson 03) guarding a relationship.
  4. TransferTo shows why aggregation matters: the same Student object moves between rooms — nobody copies students; the reference moves.
